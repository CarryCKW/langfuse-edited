部署到 k8s，修改完后，在你的 k8s 集群节点上执行：

# 确保 kubectl 能访问你的集群
kubectl cluster-info

# 部署
kubectl apply -f langfuse-k8s-deployment.yaml

# 查看部署状态
kubectl get pods -n langfuse

# 查看日志
kubectl logs -f deployment/langfuse-web -n langfuse

# 删除任务
kubectl delete -f langfuse-k8s-deployment.yaml

#常用查看命令
# 查看 pod 状态
kubectl get pods -n langfuse

# 查看 service
kubectl get svc -n langfuse

# 查看日志
kubectl logs -f deployment/langfuse-web -n langfuse
kubectl logs -f deployment/langfuse-worker -n langfuse

# 进入 pod 调试
kubectl exec -it deployment/langfuse-web -n langfuse -- /bin/sh
